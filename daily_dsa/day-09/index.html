TITLE OF PROBLEM

Given an integer, write a function to determine if it is a power of three.

Input : Integer

Output : Boolean value

Example

Example 1

Input: 0

Output: False

Example 2

Input: 9

Output: True

Example 3

Input: 100

Output: False

Challenge

Could you do it without using any loop/recursion ?

Implement

C/C++

/**
 *  Power of Three
 */

class Solution {
    public:
        bool isPowerOfThree(int n) {
            return isPowerOfThree_03(n);    // 140ms
            return isPowerOfThree_02(n);    // 130ms
            return isPowerOfThree_01(n);    // 140ms
            return isPowerOfThree_recur(n); // 136ms
            return isPowerOfThree_loop(n);  // 168ms
        }
        void init(unordered_map<int, bool> &power) {
            int p = 1;
            power[1] = true;
            while (1) {
                p *= 3;
                power[p] = true;
                if (p > INT_MAX / 3) break;
            }
        }

        bool isPowerOfThree_03(int n) {
            double logRes = log10(n) / log10(3);
            return (logRes - int(logRes) == 0);
        }
        bool isPowerOfThree_02(int n) {
            return n > 0 ? (1162261467 % n == 0) : false;
        }
        bool isPowerOfThree_01(int n) {
            static unordered_map<int, bool> power;
            if (power.size() == 0) init(power);
            return power.find(n) != power.end();
        }
        bool isPowerOfThree_recur(int n) {
            if (n == 1 || n == 3) return true;
            if (n == 0 || n % 3 != -) return false;
            return isPowerOfThree_recur(n/3);
        }
        bool isPowerOfThree_loop(int n) {
            for (; n > 0; n /= 3) {
                if (n == 1 || n == 3) return true;
                if (n == 0 || n % 3 != 0) return false;
            }
            return false;
        }
}
